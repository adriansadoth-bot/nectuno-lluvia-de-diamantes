        function inicializarEstrellas() {
            starsArray = [];
            const numeroEstrellas = Math.floor((starsCanvas.width * starsCanvas.height) / 3000);
            for (let i = 0; i < numeroEstrellas; i++) {
                starsArray.push({
                    x: Math.random() * starsCanvas.width,

                    size: Math.random() * 1.5 + 0.5,
                    blinkSpeed: Math.random() * 0.02 + 0.005,
                    alpha: Math.random()
                });
            }
        }

        function animarEstrellas() {
            ctxStars.clearRect(0, 0, starsCanvas.width, starsCanvas.height);
            ctxStars.fillStyle = 'rgba(255, 255, 255, ';
            for (let i = 0; i < starsArray.length; i++) {
                let star = starsArray[i];
                star.alpha += star.blinkSpeed;
                if (star.alpha > 1 || star.alpha < 0) {
                    star.blinkSpeed = -star.blinkSpeed;
                }
                ctxStars.fillStyle = `rgba(255, 255, 255, ${Math.max(0.1, Math.min(star.alpha, 1))})`;
                ctxStars.beginPath();
                ctxStars.arc(star.x, star.y, star.size, 0, Math.PI * 2);
                ctxStars.fill();
            }
            requestAnimationFrame(animarEstrellas);
        }

        class ParticulaMolecular {
            constructor(tipo) {
                this.tipo = tipo; // 'metano', 'carbono_libre', 'diamante', 'hidrogeno'
                this.x = Math.random() * simCanvas.width;
                // Ajustar altura inicial dependiendo de si es precipitación (paso 6)
                this.y = tipo === 'diamante' && currentPasoGuiado === 6 ? Math.random() * (simCanvas.height * 0.3) : Math.random() * simCanvas.height;
                this.vx = (Math.random() - 0.5) * 2;
                this.vy = (Math.random() - 0.5) * 2;
                this.radius = tipo === 'metano' ? 12 : tipo === 'carbono_libre' ? 6 : tipo === 'diamante' ? 8 : 4;
                this.color = this.obtenerColor();
            }

            obtenerColor() {
                switch(this.tipo) {
                    case 'metano': return '#00d2ff';
                    case 'carbono_libre': return '#555555';
                    case 'diamante': return '#e0ffff';
                    case 'hidrogeno': return '#ff6b6b';
                }
            }

            actualizar(speedMod, tempMod) {
                // Modificador de velocidad basado en temperatura y lógica de pasos
                let baseSpeed = tempMod * 1.5;
                
                if (this.tipo === 'diamante' && currentPasoGuiado === 6) {
                    // Caída o precipitación vertical suave hacia el núcleo
                    this.vy = Math.abs(this.vy) * 0.4 + 1.2;
                    this.vx *= 0.95;
                } else {
                    this.x += this.vx * baseSpeed;
                    this.y += this.vy * baseSpeed;
                }

                // Rebotes cinéticos perimetrales
                if (this.x - this.radius < 0 || this.x + this.radius > simCanvas.width) this.vx *= -1;
                if (this.y - this.radius < 0 || this.y + this.radius > simCanvas.height) this.vy *= -1;

                // Límites de caída para paso 6
                if (this.y > simCanvas.height && currentPasoGuiado === 6) {
                    this.y = 0;
                    this.x = Math.random() * simCanvas.width;
                }
            }

            dibujar() {
                ctxSim.beginPath();
                if (this.tipo === 'metano') {
                    // Estructura simulada con núcleo de Carbono y enlaces periféricos de Hidrógeno
                    ctxSim.fillStyle = 'rgba(0, 210, 255, 0.2)';
                    ctxSim.arc(this.x, this.y, this.radius, 0, Math.PI*2);
                    ctxSim.fill();
                    ctxSim.fillStyle = '#ffffff';
                    ctxSim.beginPath();
                    ctxSim.arc(this.x, this.y, 4, 0, Math.PI*2);
                    ctxSim.fill();
                } else if (this.tipo === 'diamante') {
                    // Geometría angular cristalina romboidal
                    ctxSim.fillStyle = '#ffffff';
                    ctxSim.shadowColor = '#00ffff';
                    ctxSim.shadowBlur = 8;
                    ctxSim.moveTo(this.x, this.y - this.radius);
                    ctxSim.lineTo(this.x + this.radius, this.y);
                    ctxSim.lineTo(this.x, this.y + this.radius);
                    ctxSim.lineTo(this.x - this.radius, this.y);
                    ctxSim.closePath();
                    ctxSim.fill();
                    ctxSim.shadowBlur = 0; // Reset
                } else {
                    // Partículas esféricas estándar para elementos atómicos simples
                    ctxSim.fillStyle = this.color;
                    ctxSim.arc(this.x, this.y, this.radius, 0, Math.PI*2);
                    ctxSim.fill();
                }
            }
        }

                sincronizarContenidosPaso();
            } else {
                currentPasoGuiado = 1;
                sincronizarContenidosPaso();
            }

            actualizarHUD();
        }

        function sincronizarContenidosPaso() {
            document.getElementById('step-title-display').innerText = textosPasos[currentPasoGuiado].title;
            document.getElementById('step-desc-display').innerText = textosPasos[currentPasoGuiado].desc;
            generarPoblaciónParticulas();
            actualizarHUD();
        }

        function cambiarPasoGuiado(direccion) {
            currentPasoGuiado += direccion;
            if(currentPasoGuiado < 1) currentPasoGuiado = 1;
            if(currentPasoGuiado > 6) currentPasoGuiado = 6;

            // Mapear paso guiado a la posición aproximada de los deslizadores de control físico
            if(currentPasoGuiado === 1 || currentPasoGuiado === 2) {
                document.getElementById('slide-presion').value = 1;
                document.getElementById('slide-temp').value = 1;
            } else if (currentPasoGuiado === 3 || currentPasoGuiado === 4) {
                document.getElementById('slide-presion').value = 2;
                document.getElementById('slide-temp').value = 2;
            } else {
                document.getElementById('slide-presion').value = 3;
                document.getElementById('slide-temp').value = 3;
            }

            // Sincronizar selectores textuales intermedios
            const lblP = ["Baja", "Media", "Extrema"];
            const lblT = ["Fría", "Alta", "Extrema"];
            presionNivel = parseInt(document.getElementById('slide-presion').value);
            temperaturaNivel = parseInt(document.getElementById('slide-temp').value);
            document.getElementById('val-presion').innerText = lblP[presionNivel-1];
            document.getElementById('val-temperatura').innerText = lblT[temperaturaNivel-1];

            sincronizarContenidosPaso();
        }

        function iniciarSimulacion() {
            simRunning = true;
            document.getElementById('btn-start').disabled = true;
            document.getElementById('btn-pause').disabled = false;
            if(particulas.length === 0) generarPoblaciónParticulas();
            actualizarHUD();
        }

        function pausarSimulacion() {
            simRunning = false;
            document.getElementById('btn-start').disabled = false;
            document.getElementById('btn-pause').disabled = true;
            actualizarHUD();
        }

        function reiniciarSimulacion() {
            currentPasoGuiado = 1;
            document.getElementById('slide-presion').value = 1;
            document.getElementById('slide-temp').value = 1;
            presionNivel = 1;
            temperaturaNivel = 1;
            document.getElementById('val-presion').innerText = "Baja";
            document.getElementById('val-temperatura').innerText = "Fría";
            
            sincronizarContenidosPaso();
            iniciarSimulacion();
        }

        function loopSimulador() {
            ctxSim.clearRect(0, 0, simCanvas.width, simCanvas.height);

            // Dibujar cuadrícula de fondo simulando interfaz de ordenador científico
            ctxSim.strokeStyle = 'rgba(0, 210, 255, 0.03)';
            ctxSim.lineWidth = 1;
            const sizeCuadro = 30;
            for(let x=0; x<simCanvas.width; x+=sizeCuadro) {
                ctxSim.beginPath(); ctxSim.moveTo(x, 0); ctxSim.lineTo(x, simCanvas.height); ctxSim.stroke();
            }
            for(let y=0; y<simCanvas.height; y+=sizeCuadro) {
                ctxSim.beginPath(); ctxSim.moveTo(0, y); ctxSim.lineTo(simCanvas.width, y); ctxSim.stroke();
            }

            // Actualización modular de partículas físicas
            for(let i=0; i<particulas.length; i++) {
                if (simRunning) {
                    particulas[i].actualizar(presionNivel, temperaturaNivel);
                }
                particulas[i].dibujar();
            }

            requestAnimationFrame(loopSimulador);
        }

        let quizIndexActual = 0;

        function cargarPreguntaQuiz() {
            const feedbackBox = document.getElementById('quiz-feedback-box');
            const btnNext = document.getElementById('quiz-btn-next');
            
            feedbackBox.style.display = 'none';
            btnNext.style.display = 'none';

            if(quizIndexActual >= bancoPreguntas.length) {
                mostrarResultadosFinalesQuiz();
                return;
            }

            // Progreso lineal visual
            const pctProgreso = (quizIndexActual / bancoPreguntas.length) * 100;
            document.getElementById('quiz-progress-inner').style.width = `${pctProgreso}%`;

            const itemPregunta = bancoPreguntas[quizIndexActual];
            document.getElementById('quiz-txt-pregunta').innerText = `${quizIndexActual + 1}. ${itemPregunta.q}`;

            const bloqueOpciones = document.getElementById('quiz-opciones-bloque');
            bloqueOpciones.innerHTML = '';

            itemPregunta.options.forEach((opcionText, index) => {
                const btn = document.createElement('button');
                btn.className = 'option-btn';
                btn.innerText = opcionText;
                btn.onclick = () => verificarRespuestaQuiz(index, btn);
                bloqueOpciones.appendChild(btn);
            });
        }

        function verificarRespuestaQuiz(indexSeleccionado, botonElemento) {
            const itemPregunta = bancoPreguntas[quizIndexActual];
            const todosLosBotones = document.querySelectorAll('#quiz-opciones-bloque .option-btn');
            
            // Bloquear interacciones extras de clics secundarios
            todosLosBotones.forEach(b => b.disabled = true);

            const feedbackBox = document.getElementById('quiz-feedback-box');
            feedbackBox.style.display = 'block';
            feedbackBox.innerText = itemPregunta.feedback;

            if(indexSeleccionado === itemPregunta.correct) {
                botonElemento.classList.add('correct');
                feedbackBox.style.background = 'rgba(16, 185, 129, 0.1)';
                feedbackBox.style.border = '1px solid #10b981';
                feedbackBox.style.color = '#34d399';
                quizPuntajeAcumulado++;
            } else {
                botonElemento.classList.add('incorrect');
                todosLosBotones[itemPregunta.correct].classList.add('correct');
                feedbackBox.style.background = 'rgba(239, 68, 68, 0.1)';
                feedbackBox.style.border = '1px solid #ef4444';
                feedbackBox.style.color = '#f87171';
            }

            document.getElementById('quiz-btn-next').style.display = 'block';
        }

        function avanzarQuiz() {
            quizIndexActual++;
            cargarPreguntaQuiz();
        }
